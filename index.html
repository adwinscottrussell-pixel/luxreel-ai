const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', } const TONES = { luxury: 'Write a sophisticated Instagram caption (2-3 lines, max 180 chars) for a luxury brand. End with 3-4 premium hashtags. No emojis. Refined, elegant, aspirational.', bold: 'Write a punchy, confident Instagram caption (1-2 lines). Empowering, direct tone. 3-4 power hashtags. Short and impactful.', minimal: 'Write a minimalist Instagram caption. ONE short sentence only. Max 80 characters. Ultra-clean, no filler. 2 hashtags.', playful: 'Write a fun, relatable Instagram caption. 2-3 lines. Light humor. 4-5 trending hashtags with 2-3 fitting emojis.', cinematic: 'Write a cinematic, poetic Instagram caption. 2-3 evocative lines, like a film title card. Atmospheric. 3 hashtags.', } Deno.serve(async (req) => { if (req.method === 'OPTIONS') { return new Response('ok', { headers: corsHeaders }) } let prompt = 'a cinematic luxury scene' let tone = 'luxury' let action = 'start' // 'start' or 'poll' let taskId = '' try { const body = await req.json() prompt = body.prompt || prompt tone = body.tone || tone action = body.action || 'start' taskId = body.taskId || '' } catch(e) { console.warn('Body parse error:', String(e)) } const runwayKey = Deno.env.get('RUNWAY_API_KEY') || '' const openaiKey = Deno.env.get('OPENAI_API_KEY') || '' // ── POLL ACTION — check if Runway task is done ── if (action === 'poll' && taskId) { console.log('Polling task:', taskId) try { const res = await fetch(`https://api.dev.runwayml.com/v1/tasks/${taskId}`, { headers: { 'Authorization': `Bearer ${runwayKey}`, 'X-Runway-Version': '2024-11-06', }, }) const data = await res.json() console.log('Full task response:', JSON.stringify(data)) console.log('Poll result:', data.status) if (data.status === 'SUCCEEDED') { // Try all possible output field names const videoUrl = data.output?.[0] || data.artifacts?.[0]?.url || data.url || data.video_url || data.videoUrl || '' console.log('Video URL found:', videoUrl) return new Response( JSON.stringify({ status: 'done', videoUrl }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } } ) } if (data.status === 'FAILED') { console.error('Task failed:', JSON.stringify(data)) return new Response( JSON.stringify({ status: 'failed', videoUrl: '' }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } } ) } // Still running return new Response( JSON.stringify({ status: 'running', videoUrl: '' }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } } ) } catch(e) { console.error('Poll error:', String(e)) return new Response( JSON.stringify({ status: 'error', videoUrl: '' }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } } ) } } // ── START ACTION — kick off caption + video task ── console.log('Starting — prompt:', prompt, 'tone:', tone) // 1. Generate caption let caption = '' if (openaiKey) { try { const r = await fetch('https://api.openai.com/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${openaiKey}`, }, body: JSON.stringify({ model: 'gpt-4o-mini', messages: [ { role: 'system', content: TONES[tone] || TONES.luxury }, { role: 'user', content: `Video concept: "${prompt}"` }, ], max_tokens: 160, temperature: 0.85, }), }) const d = await r.json() caption = d.choices?.[0]?.message?.content?.trim() || '' console.log('Caption:', caption.substring(0, 60)) } catch(e) { console.warn('Caption error:', String(e)) } } // 2. Start Runway task — return immediately with task ID let runwayTaskId = '' if (runwayKey) { try { const enhancedPrompt = `${prompt}. Cinematic quality, smooth camera movement, professional lighting, luxury aesthetic.` console.log('Starting Runway task...') const taskRes = await fetch('https://api.dev.runwayml.com/v1/text_to_video', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${runwayKey}`, 'X-Runway-Version': '2024-11-06', }, body: JSON.stringify({ model: 'gen4.5', promptText: enhancedPrompt, ratio: '1280:720', duration: 5, }), }) const taskData = await taskRes.json() console.log('Runway response:', JSON.stringify(taskData)) if (taskData.id) { runwayTaskId = taskData.id console.log('Task ID:', runwayTaskId) } else { console.error('No task ID:', JSON.stringify(taskData)) } } catch(e) { console.error('Runway error:', String(e)) } } // Return immediately with caption + task ID // App will poll separately using the task ID return new Response( JSON.stringify({ caption, taskId: runwayTaskId, status: runwayTaskId ? 'pending' : 'failed' }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } } ) })
